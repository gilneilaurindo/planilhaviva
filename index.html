/***** PlanilhaViva: backend (Google Apps Script) *****/ const CLIENT_ID = "599756107964-ob0m7i1asuc0t7ng7rn6qc1106elf2ou.apps.googleusercontent.com"; const SHEET_TX = "Lançamentos"; const SHEET_USERS = "Usuários"; const TX_HEADERS = ["ID", "Email", "Data", "Descrição", "Tipo", "Categoria", "Valor", "Criado em"]; const USER_HEADERS = ["Email", "Nome", "Primeiro acesso", "Último acesso"]; const CATEGORIES = ["Receita", "Dízimo", "Fixos", "Pessoal", "Saúde", "Dívidas", "Investimento", "Outros"]; const MAX_AMOUNT = 99999999.99; class AppError extends Error { constructor(message, code) { super(message); this.code = code || "APP"; } } function doGet() { return json({ status: "success", message: "PlanilhaViva online" }); } function doPost(e) { try { const req = JSON.parse(e.postData.contents || "{}"); const user = verifyGoogleToken(req.token); switch (req.action) { case "login": return json(registerUser(user)); case "add_transaction": return json(addTransaction(user, req)); case "list_transactions": return json(listTransactions(user)); case "summary": return json(summary(user)); case "count_users": return json(countUsers()); case "delete_my_data": return json(deleteUserData(user)); default: throw new AppError("Ação inválida."); } } catch (err) { if (err instanceof AppError && err.code === "AUTH") return json({ status: "error", code: "AUTH", message: err.message }); if (err instanceof AppError) return json({ status: "error", message: err.message }); console.error(err); return json({ status: "error", message: "Erro interno. Tente novamente em instantes." }); } } /* ---------- Segurança: valida o token do Google no servidor ---------- */ function verifyGoogleToken(token) { if (!token || typeof token !== "string" || token.length > 4096) throw new AppError("Sessão ausente. Entre novamente.", "AUTH"); const res = UrlFetchApp.fetch( "https://oauth2.googleapis.com/tokeninfo?id_token=" + encodeURIComponent(token), { muteHttpExceptions: true } ); if (res.getResponseCode() !== 200) throw new AppError("Sessão inválida. Entre novamente.", "AUTH"); const info = JSON.parse(res.getContentText()); if (info.aud !== CLIENT_ID) throw new AppError("Sessão inválida. Entre novamente.", "AUTH"); if (Number(info.exp) * 1000 < Date.now()) throw new AppError("Sessão expirada. Entre novamente.", "AUTH"); if (String(info.email_verified) !== "true") throw new AppError("E-mail da conta Google não verificado.", "AUTH"); return { email: String(info.email).toLowerCase(), name: info.name || info.given_name || "" }; } /* ---------- Utilitários ---------- */ function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function getSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold"); sh.setFrozenRows(1); } return sh; } // Evita injeção de fórmulas na planilha (ex.: "=IMPORTRANGE(...)") function safeText(v) { const s = String(v).trim().slice(0, 80); return /^[=+\-@]/.test(s) ? "'" + s : s; } function parseIsoDate(s) { const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(s || "")); if (!m) return null; const d = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])); return isNaN(d.getTime()) ? null : d; } function formatDate(d) { return d instanceof Date ? Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd") : String(d); } const round2 = v => Math.round(v * 100) / 100; /* ---------- Usuários ---------- */ function registerUser(user) { const sh = getSheet(SHEET_USERS, USER_HEADERS); const data = sh.getDataRange().getValues(); const now = new Date(); let found = false; for (let i = 1; i < data.length; i++) { if (String(data[i][0]).toLowerCase() === user.email) { sh.getRange(i + 1, 2).setValue(user.name); sh.getRange(i + 1, 4).setValue(now); found = true; break; } } if (!found) sh.appendRow([user.email, user.name, now, now]); return { status: "success", total: Math.max(0, sh.getLastRow() - 1) }; } function countUsers() { const sh = getSheet(SHEET_USERS, USER_HEADERS); return { status: "success", total: Math.max(0, sh.getLastRow() - 1) }; } /* ---------- Lançamentos ---------- */ function addTransaction(user, req) { const desc = String(req.desc || "").trim(); const amount = round2(Number(req.amount)); const type = req.type === "in" ? "Entrada" : req.type === "out" ? "Saída" : ""; const category = CATEGORIES.indexOf(req.category) >= 0 ? req.category : ""; const date = parseIsoDate(req.date); if (desc.length < 2 || desc.length > 80) throw new AppError("A descrição deve ter entre 2 e 80 caracteres."); if (!(amount > 0) || amount > MAX_AMOUNT) throw new AppError("Valor inválido."); if (!type) throw new AppError("Natureza do lançamento inválida."); if (!category) throw new AppError("Categoria inválida."); if (!date) throw new AppError("Data inválida."); const sh = getSheet(SHEET_TX, TX_HEADERS); sh.appendRow([Utilities.getUuid(), user.email, date, safeText(desc), type, category, amount, new Date()]); return { status: "success", message: "Lançamento gravado." }; } function readRowsFor(email) { const sh = getSheet(SHEET_TX, TX_HEADERS); return sh.getDataRange().getValues().slice(1) .filter(r => String(r[1]).toLowerCase() === email) .map(r => ({ id: String(r[0]), date: formatDate(r[2]), desc: String(r[3]).replace(/^'/, ""), type: r[4] === "Entrada" ? "in" : "out", category: String(r[5]), amount: Number(r[6]) })); } function listTransactions(user) { const rows = readRowsFor(user.email) .sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0)) .slice(0, 30); return { status: "success", lancamentos: rows }; } function summary(user) { const rows = readRowsFor(user.email); let entradas = 0, saidas = 0, investido = 0; const gastosPorCategoria = {}; rows.forEach(r => { if (r.category === "Investimento") investido += r.amount; if (r.type === "in") { entradas += r.amount; } else { saidas += r.amount; gastosPorCategoria[r.category] = round2((gastosPorCategoria[r.category] || 0) + r.amount); } }); return { status: "success", entradas: round2(entradas), saidas: round2(saidas), saldo: round2(entradas - saidas), investido: round2(investido), gastosPorCategoria: gastosPorCategoria, quantidade: rows.length }; } /* ---------- Exclusão de dados (LGPD: direito de eliminação) ---------- */ function deleteUserData(user) { const tx = getSheet(SHEET_TX, TX_HEADERS); const txRows = tx.getDataRange().getValues(); for (let i = txRows.length - 1; i >= 1; i--) { if (String(txRows[i][1]).toLowerCase() === user.email) tx.deleteRow(i + 1); } const us = getSheet(SHEET_USERS, USER_HEADERS); const uRows = us.getDataRange().getValues(); for (let i = uRows.length - 1; i >= 1; i--) { if (String(uRows[i][0]).toLowerCase() === user.email) us.deleteRow(i + 1); } return { status: "success", message: "Seus dados foram excluídos." }; }
